       ```python
       def read_guid(self) -> UUID:
           """读取 FGuid（16 字节）。
           
           UE FGuid 序列化为 4 个 uint32，按特定顺序排列。
           Source: Runtime/Core/Public/Misc/Guid.h
           
           FGuid serialization format:
           - uint32 A (4 bytes)
           - uint32 B (4 bytes)
           - uint32 C (4 bytes)
           - uint32 D (4 bytes)
           Total: 16 bytes
           
           Returns:
               UUID 对象（标准格式）
           """
           import uuid
           # FGuid 序列化：A, B, C, D 各 4 字节
           a = self.read_uint32()
           b = self.read_uint32()
           c = self.read_uint32()
           d = self.read_uint32()
           
           # 转换为 UUID 格式（使用 fields 构造）
           # UUID fields: time_low, time_mid, time_hi_version, clock_seq_hi_variant, clock_seq_low, node
           # FGuid A,B,C,D 对应不同的 UUID 构件
           return uuid.UUID(fields=(
               a & 0xFFFFFFFF,           # time_low (from A)
               (b >> 16) & 0xFFFF,       # time_mid (from B high)
               (b & 0xFFFF),             # time_hi_version (from B low)
               (c >> 24) & 0xFF,         # clock_seq_hi_variant (from C high)
               (c >> 16) & 0xFF,         # clock_seq_low (from C mid-high)
               d                         # node (from D)
           ))
       ```
       
    2. 定义 CustomVersionEntry dataclass：

       ```python
       @dataclass
       class CustomVersionEntry:
           """单个 CustomVersion 条目。
           
           对应 UE FCustomVersion 结构。
           Source: Runtime/Core/Public/Serialization/CustomVersion.h
           """
           guid: UUID                      # 模块 GUID
           name: str                       # 模块名称（从 CUSTOM_VERSION_GUIDS 映射）
           version: int                    # 版本号
           
           __source: str = "CustomVersion.h"  # 源码追踪
       ```
       
    3. 定义 CustomVersionContainer 类：
       ```python
       class CustomVersionContainer:
           """CustomVersion 容器解析器。
           
           对应 UE FCustomVersionContainer，存储各模块的版本号。
           Source: Runtime/Core/Public/Serialization/CustomVersion.h
           
           使用 PackageFileSummary 提供的 custom_version_offset 定位数据。
           """
           
           @staticmethod
           def parse(archive: UAssetArchive, summary: PackageFileSummary) -> list[CustomVersionEntry]:
               """解析 CustomVersionContainer。
               
               Args:
                   archive: UAssetArchive 实例
                   summary: PackageFileSummary（包含 custom_version_count 和 custom_version_offset）
                   
               Returns:
                   CustomVersionEntry 列表
                   
               **关键修复（per checker）：使用 summary.custom_version_offset seek 到正确位置**
               
               CustomVersionContainer 序列化格式：
               - 条目数组：每个条目包含 GUID(16 bytes) + Version(4 bytes)
               - 总大小 = count * 20 bytes
               
               Source: Runtime/Core/Private/Serialization/CustomVersion.cpp
               """
               entries = []
               
               # 使用 summary 提供的偏移定位数据（per checker 修复）
               if summary.custom_version_count == 0:
                   return entries
               
               # Seek 到条目数组起始位置（不含 count 字段）
               archive.seek(summary.custom_version_offset)
               
               # 读取每个条目
               for _ in range(summary.custom_version_count):
                   guid = archive.read_guid()
                   version = archive.read_int32()
                   
                   # 映射 GUID 到名称（per D-06）
                   guid_str = str(guid).upper()
                   name = get_custom_version_name(guid_str)
                   
                   entry = CustomVersionEntry(
                       guid=guid,
                       name=name,
                       version=version
                   )
                   entries.append(entry)
               
               return entries
       ```
