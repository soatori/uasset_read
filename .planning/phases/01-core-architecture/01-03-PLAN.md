    2. 实现 parse 方法核心逻辑（对应 UE LinkerLoad.cpp，关键修复：保存 CustomVersion 偏移）：
       ```python
       def parse(self, archive: UAssetArchive) -> PackageFileSummary:
           # 读取 Tag（已在 archive.open() 中验证）
           archive.seek(0)
           tag = archive.read_int32()  # offset 0
           
           # 读取双版本号（UE5 FPackageFileVersion）
           version_ue4 = archive.read_int32()  # offset 4
           version_ue5 = archive.read_int32()  # offset 8
           version = FPackageFileVersion(version_ue4, version_ue5)
           
           # 设置 archive 的版本（供后续版本感知读取）
           archive.set_version(version)
           
           # 版本检查（per D-14：版本不支持时尝试继续并警告）
           if version_ue4 < VER_UE4_OLDEST_LOADABLE_PACKAGE:
               log_warn(f"版本过低: {version_ue4} < {VER_UE4_OLDEST_LOADABLE_PACKAGE}, 尝试继续解析")
           
           # 读取 LicenseeVersion（offset 12，UE4 < 215）
           # 或 CustomVersionContainer 后（UE4 >= 215）
           
           # === 关键修复：保存 CustomVersion 偏移供 Plan 04 使用 ===
           custom_version_count = 0
           custom_version_offset = 0
           
           if version_ue4 >= 215:  # VER_UE4_ADDED_CUSTOM_VERSIONS
               # 读取 CustomVersionContainer
               custom_version_offset = archive.tell()  # 记录数据偏移（跳过 count 字段后的位置）
               custom_version_count = archive.read_int32()  # 读取条目数量
               
               # 记录实际条目数组的起始偏移（供 Plan 04 seek 使用）
               custom_version_data_offset = archive.tell()
               
               # 跳过 CustomVersion 条目区域
               # 每个条目 = FGuid(16 bytes) + Version(4 bytes) = 20 bytes
               skip_size = custom_version_count * 20
               archive.seek(custom_version_data_offset + skip_size)
               
               # 更新 custom_version_offset 为条目数组起始位置（不含 count 字段）
               custom_version_offset = custom_version_data_offset
           
           # 读取 LicenseeVersion（在 CustomVersionContainer 之后）
           licensee_version = archive.read_int32()
           
           # 读取 PackageFlags
           package_flags = archive.read_uint32()
           
           # 读取 TotalHeaderSize
           total_header_size = archive.read_int32()
           
           # 读取 PackageName (FString) - 使用 archive.read_fstring()
           # per checker 修复：使用 Archive 方法而非 parser 私有方法
           package_name = archive.read_fstring()
           
           # 读取 NameCount/NameOffset
           name_count = archive.read_int32()
           name_offset = archive.read_int32()
           
           # 读取 ExportCount/ExportOffset
           export_count = archive.read_int32()
           export_offset = archive.read_int32()
           
           # 读取 ImportCount/ImportOffset
           import_count = archive.read_int32()
           import_offset = archive.read_int32()
           
           # 读取 DependsOffset
           depends_offset = archive.read_int32()
           
           # 读取 SoftObjectPaths（UE5）
           soft_object_paths_count = 0
           soft_object_paths_offset = 0
           if version_ue5 >= 1003:  # ADD_SOFTOBJECTPATH_LIST
               soft_object_paths_count = archive.read_int32()
               soft_object_paths_offset = archive.read_int32()
           
           # 读取 BulkDataStartOffset
           bulk_data_start_offset = archive.read_int64()
           
           # 读取 PayloadTocOffset（UE5）
           payload_toc_offset = 0
           if version_ue5 >= 1002:  # PAYLOAD_TOC
               payload_toc_offset = archive.read_int64()
           
           # 构建结果（包含 custom_version_count 和 custom_version_offset）
           return PackageFileSummary(
               tag=tag,
               version=version,
               licensee_version=licensee_version,
               custom_version_count=custom_version_count,      # 新增：供 Plan 04 使用
               custom_version_offset=custom_version_offset,    # 新增：供 Plan 04 使用
               package_flags=package_flags,
               total_header_size=total_header_size,
               package_name=package_name,
               name_count=name_count,
               name_offset=name_offset,
               export_count=export_count,
               export_offset=export_offset,
               import_count=import_count,
               import_offset=import_offset,
               depends_offset=depends_offset,
               bulk_data_start_offset=bulk_data_start_offset,
               payload_toc_offset=payload_toc_offset,
               # ... 其他字段
           )
       ```
