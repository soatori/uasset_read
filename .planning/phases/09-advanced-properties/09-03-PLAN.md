    def test_struct_property_basic():
        """测试 StructProperty 基本解析。"""
        name_map = ["TestStruct", "None", "X", "Y", "Z"]

        # 构造 Mock PropertyTag 和数据
        tag = PropertyTag(
            name="TestStruct",
            type="StructProperty(/Script/CoreUObject.Vector)",
            size=0
        )

        # 构造 Mock 数据：PropertyTag 循环
        # Name="X" (IntProperty) + Value=1.0

            struct.pack('<i', 4) + struct.pack('<B', 0) +  # Size + Flags
            struct.pack('<f', 1.0) +  # Value
            # Y 字段
            struct.pack('<I', 3) + struct.pack('<I', 0) +  # FName: "Y"
            struct.pack('<i', 12) + b"FloatProperty\x00" +
            struct.pack('<i', 4) + struct.pack('<B', 0) +
            struct.pack('<f', 2.0) +
            # Z 字段

            struct.pack('<i', 4) + struct.pack('<B', 0) +
            struct.pack('<f', 3.0) +
            # None 结束标记
            struct.pack('<I', 1) + struct.pack('<I', 0) +  # FName: "None"
        )

        archive = create_mock_archive_with_data(data)
        summary = create_mock_summary()
        export_map = []

        result = parse_struct_property(tag, archive, name_map, export_map, summary, depth=0)

        assert isinstance(result, StructValue)
        assert result.property_type == "StructProperty"
        assert result.struct_type == "/Script/CoreUObject.Vector"
        assert "X" in result.fields
        assert "Y" in result.fields
        assert "Z" in result.fields

    def test_struct_property_depth_limit():
