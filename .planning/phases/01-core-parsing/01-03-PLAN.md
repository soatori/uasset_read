<objective>
Fix SavedHash parsing bug discovered during UAT testing with real Lyra UE5 files.

Purpose: Real UE5 files with UE5Version >= 1004 contain SavedHash (20 bytes) and early TotalHeaderSize (4 bytes) before CustomVersions. Current parser misses these fields, causing all subsequent field reads to be offset by 24 bytes, resulting in garbage values for NameOffset/ImportOffset/ExportOffset.

Output: Corrected PackageFileSummary dataclass and read_package_summary() that properly handles PACKAGE_SAVED_HASH version threshold.
</objective>

    ```python
    def test_saved_hash_ue5_package_saved_hash_version():
        """Test SavedHash and early TotalHeaderSize parsing for UE5 >= PACKAGE_SAVED_HASH (1004)."""
        # Create synthetic data with UE5 version 1004 (PACKAGE_SAVED_HASH)
        data = bytearray()
        
        # Header fields in correct order:
        # 1. Magic tag (0x9E2A83C1)
        data.extend(struct.pack('<I', 0x9E2A83C1))
        
        # 2. LegacyFileVersion (-8 for UE5)
        data.extend(struct.pack('<i', -8))
        
        # 3. LegacyUE3Version (skipped for legacy=-8)
        
        # 4. UE4 version
        data.extend(struct.pack('<i', 1000))
        
        # 5. UE5 version (1004 = PACKAGE_SAVED_HASH)
        data.extend(struct.pack('<i', 1004))
        
        # 6. LicenseeVersion
        data.extend(struct.pack('<i', 0))
        
        # 7. SavedHash (20 bytes) - NEW for UE5 >= 1004
        test_saved_hash = bytes([0xAA] * 20)  # Test pattern
        data.extend(test_saved_hash)
        
        # 8. TotalHeaderSize (early read) - NEW for UE5 >= 1004
        data.extend(struct.pack('<i', 256))
        
        # 9. CustomVersions (count = 0)
        data.extend(struct.pack('<I', 0))
        
        # 10. PackageFlags
        data.extend(struct.pack('<I', 0))
        
        # 11. NameCount + NameOffset
        data.extend(struct.pack('<i', 0))  # NameCount
        data.extend(struct.pack('<i', len(data) + 8))  # NameOffset
        
        # Pad to reasonable size
        while len(data) < 300:
            data.append(0)
        
        # Write and parse
        with tempfile.NamedTemporaryFile(suffix='.uasset', delete=False) as f:
            f.write(data)
            temp_path = f.name
        
        try:
            result = parse_uasset(temp_path)

            assert result.summary.saved_hash == test_saved_hash, \
                f"SavedHash mismatch: expected {test_saved_hash.hex()}, got {result.summary.saved_hash.hex()}"
            assert result.summary.total_header_size == 256, \
                f"TotalHeaderSize mismatch: expected 256, got {result.summary.total_header_size}"
        finally:
            os.unlink(temp_path)
    ```
