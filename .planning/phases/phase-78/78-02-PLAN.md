**问题**：当前 parse_uasset_with_linker 中，PackageLinker 共享 parse_uasset 创建的同一个 FArchive 实例。如果连续调用 parse_uasset_with_linker(file_A) 和 parse_uasset_with_linker(file_B)，archive 状态可能串扰。

**方案**：修改 parse_uasset_with_linker 为 linker 创建独立的 FArchive 实例，并把该实例的生命周期移交给 `LinkerParseResult`。

修改 src/uasset_read/parse_uasset.py 中的 parse_uasset_with_linker 函数：

```python
def parse_uasset_with_linker(
    path: str,
    tolerant: bool = True,
    preload_all: bool = False,
) -> "LinkerParseResult":
    from uasset_read.link.linker import PackageLinker

    result = LinkerParseResult()
    archive = None
    linker_archive = None  # ← 新增：linker 专属 archive

    try:
        archive = FArchive(path, tolerant=tolerant)
        # ... 读取 summary, name_map, import_map, export_map ...
        # ... 解析 ExportMap 属性（使用原始 archive）...

        # 为 linker 创建独立 archive 实例（CUE4Parse FAssetArchive 模式）
        linker_archive = FArchive(path, tolerant=tolerant)
        linker_archive.set_byte_swapping(archive._byte_swapping)

        linker = PackageLinker(
            linker_archive,  # ← 使用独立 archive
            result.summary, result.name_map,
            result.import_map, result.export_map,
        )
        linker.link()
        result.linker = linker
        result.all_objects = linker._import_objects + linker._export_objects
        result.root_objects = linker._root_objects

        if preload_all:
            for i in range(len(linker._export_objects)):
                linker.preload(i)

        _post_process(..., linker=result.linker)

    # ... error handling ...
    finally:
        if archive:
            archive.close()
        # 注意：linker_archive 不在这里关闭
        # 它的生命周期由 result.close() / __exit__() 管理

    return result
```

同时更新 `LinkerParseResult`：

```python
class LinkerParseResult:
    linker: Optional[PackageLinker]

    def close(self) -> None:
        if self.linker and getattr(self.linker, "_archive", None):
            self.linker._archive.close()

    def __enter__(self):
        return self

    def __exit__(self, exc_type, exc, tb):
        self.close()
```

**问题**：当前 preload() 直接 archive.seek(instance.serial_offset)，破坏 archive 位置，影响调用者的读取状态。

**方案**：实现 save/restore 模式的 preload。

修改 src/uasset_read/link/linker.py 的 preload() 方法：

```python
def preload(self, index: int) -> None:
    """Phase 2: lazily deserialize properties for export *index*.
    
    使用 save/restore 模式保护 archive 位置（CUE4Parse FAssetArchive 模式）。
    """
    if index in self._preload_cache:
        return
    if index < 0 or index >= len(self._export_objects):
        return

    instance = self._export_objects[index]
    if instance._preloaded:
        self._preload_cache[index] = True
        return

    if instance.serial_size == 0:
        instance._preloaded = True
        self._preload_cache[index] = True
        return

    # Save/restore pattern — 不污染 archive 位置
    saved_pos = self._archive.tell()
    try:
        self._archive.seek(instance.serial_offset)
        
        from uasset_read.parsers.property_parser import (
            parse_properties_from_export,
        )
        
        exp = self._export_map[index]
        instance.serialized_properties = parse_properties_from_export(
            exp,
            self._archive,
            self._summary,
            self._name_map,
            self._export_map,
            self._import_map,
        )
        instance._preloaded = True
        self._preload_cache[index] = True
    finally:
        self._archive.seek(saved_pos)  # ← 始终恢复位置
```

同步修改 UObjectInstance.ensure_preloaded() 方法（object_instance.py），增加位置安全守卫：

```python
def ensure_preloaded(self) -> None:
    """Ensure properties are loaded (lazy load if needed).
    
    位置安全：通过 linker 的 save/restore preload 机制，
    调用此方法不会改变 archive 的读取位置。
    """
    if not self._preloaded and self.is_export and self.linker:
        exp_idx = self.package_index - 1
        self.linker.preload(exp_idx)
```
