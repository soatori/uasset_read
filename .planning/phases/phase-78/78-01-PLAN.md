创建 src/uasset_read/models/uobject.py，定义两套明确分离的模型：

1. **反射层次（reflection hierarchy）**  
   用于表达 UE `UField / UStruct / UClass / UFunction` 关系，供 `SuperField`、`Class`、`Template` 等解析使用。

2. **导出类标签（export class tags）**  
   用于把运行时导出对象粗分类到 `Graph / GraphNode / Component / Actor / BlueprintGeneratedClass` 等桶，供 formatter / graph / extractor 使用。

反射层次定义（使用 dataclass，与现有 models/ 风格一致）：

```python
from dataclasses import dataclass
from enum import Enum
from typing import Optional, List

class UObjectCategory(Enum):
    """UE 反射层次分类枚举。"""
    UObject = "UObject"
    UField = "UField"
    UEnum = "UEnum"
    UStruct = "UStruct"
    UClass = "UClass"
    UFunction = "UFunction"
    Unknown = "Unknown"

@dataclass
class UObjectBase:
    """Python 端 UObject 基类表示。
    
    注意：这不是序列化模型，而是反射层次/元数据工具类。
    运行时 UObjectInstance 仍使用 link/object_instance.py 中的版本。
    """
    name: str
    category: UObjectCategory = UObjectCategory.Unknown
    super_field_name: Optional[str] = None  # 父类名（通过 SuperIndex 解析）
    class_package: Optional[str] = None     # 所属包（如 /Script/CoreUObject）

    def is_field(self) -> bool:
        return self.category in (
            UObjectCategory.UField, UObjectCategory.UEnum,
            UObjectCategory.UStruct, UObjectCategory.UClass,
            UObjectCategory.UFunction,
        )

@dataclass
class UField(UObjectBase):
    """UField — 所有反射字段的基类。"""
    category: UObjectCategory = UObjectCategory.UField

@dataclass
class UEnum(UField):
    """UEnum — 枚举类型。"""
    category: UObjectCategory = UObjectCategory.UEnum
    enum_values: List[str] = None  # 可选：枚举值列表

@dataclass
class UStruct(UField):
    """UStruct — 结构体基类（Script/Struct/Class/Function 的父类）。"""
    category: UObjectCategory = UObjectCategory.UStruct
    children: List[str] = None  # 子类名列表（运行时填充）

@dataclass
class UClass(UStruct):
    """UClass — 类类型（BlueprintGeneratedClass 的基类）。"""
    category: UObjectCategory = UObjectCategory.UClass

@dataclass
class UFunction(UStruct):
    """UFunction — 函数类型。"""
    category: UObjectCategory = UObjectCategory.UFunction
```

```python
class ExportClassTag(Enum):
    Unknown = "Unknown"
    Graph = "Graph"
    GraphNode = "GraphNode"
    GraphPin = "GraphPin"
    BlueprintClass = "BlueprintClass"
    Component = "Component"
    Actor = "Actor"

# UE 反射类名 → UObjectCategory 映射
_REFLECTION_CATEGORY_MAP: dict[str, UObjectCategory] = {
    "BlueprintGeneratedClass": UObjectCategory.UClass,
    "Class": UObjectCategory.UClass,
    "UserDefinedStruct": UObjectCategory.UStruct,
    "ScriptStruct": UObjectCategory.UStruct,
    "UserDefinedEnum": UObjectCategory.UEnum,
    "Enum": UObjectCategory.UEnum,
    "Function": UObjectCategory.UFunction,
    "UFunction": UObjectCategory.UFunction,
    "Object": UObjectCategory.UObject,
}

_EXPORT_CLASS_TAG_MAP: dict[str, ExportClassTag] = {
    "EdGraph": ExportClassTag.Graph,
    "UberEdGraph": ExportClassTag.Graph,
    "EdGraphNode": ExportClassTag.GraphNode,
    "K2Node": ExportClassTag.GraphNode,
    "BlueprintGeneratedClass": ExportClassTag.BlueprintClass,
}

def get_reflection_category(class_name: str) -> UObjectCategory:
    """将 UE 反射类名分类到 UObject 层次结构。未知返回 Unknown。"""
    return _REFLECTION_CATEGORY_MAP.get(class_name, UObjectCategory.Unknown)

def get_export_class_tag(class_name: str) -> ExportClassTag:
    """将导出对象类名归入运行时标签桶。未知返回 Unknown。"""
    if class_name in _EXPORT_CLASS_TAG_MAP:
        return _EXPORT_CLASS_TAG_MAP[class_name]
    if class_name.endswith("Component"):
        return ExportClassTag.Component
    if class_name.endswith("Actor"):
        return ExportClassTag.Actor
    if class_name.startswith("K2Node_") or "GraphNode" in class_name:
        return ExportClassTag.GraphNode
    return ExportClassTag.Unknown
```

```python
def build_super_tree(self) -> None:
    """Resolve SuperIndex → UObjectInstance for all exports.
    
    Only exports have SuperIndex (imports reference external types).
    """
    for inst in self._export_objects:
        if inst._raw_export and not inst._raw_export.super_index.is_null:
            parent = self.resolve_package_index(inst._raw_export.super_index)
            if parent is not None:
                inst.super = parent  # 需要在 UObjectInstance 添加 super 字段
```

2. **resolve_class_ref(export_index)** — 返回 class 的 UObjectInstance：
```python
def resolve_class_ref(self, export_index: int) -> Optional[UObjectInstance]:
    """Resolve the class UObjectInstance for an export by index."""
    if not (0 <= export_index < len(self._export_objects)):
        return None
    exp = self._export_objects[export_index]
    if exp._raw_export:
        return self.resolve_package_index(exp._raw_export.class_index)
    return None
```

3. **resolve_template_ref(export_index)** — 返回 template (CDO) 的 UObjectInstance：
```python
def resolve_template_ref(self, export_index: int) -> Optional[UObjectInstance]:
    """Resolve the template UObjectInstance for an export by index."""
    if not (0 <= export_index < len(self._export_objects)):
        return None
    exp = self._export_objects[export_index]
    if exp._raw_export:
        return self.resolve_package_index(exp._raw_export.template_index)
    return None
```

4. **UObjectInstance.get_super_field_chain()** — 返回完整父类链：
```python
def get_super_field_chain(self) -> List["UObjectInstance"]:
    """Walk SuperIndex chain, return list from self to root.
    
    Returns [self, parent, grandparent, ...] until SuperIndex is null.
    Maximum 20 levels (sanity limit).
    """
    chain: List[UObjectInstance] = [self]
    current = self
    for _ in range(20):
        if current.super is None:
            break
        chain.append(current.super)
        current = current.super
    return chain
```
