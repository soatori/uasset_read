    ```python
    def read_import_map(
        archive: FArchive,
        summary: PackageFileSummary,
        name_map: List[str]
    ) -> List[ObjectImport]:
        """读取导入表（CORE-04）。

        UE5 条件字段（ObjectResource.cpp L40-85）：
        - PackageName: UEVer >= 518 且 !IsFilterEditorOnly
        - bImportOptional: UEVer >= 1003
        """
        archive.seek(summary.import_offset)
        import_map: List[ObjectImport] = []

        is_ue5_file = summary.legacy_file_version <= -8
        # UE5 文件自动满足 UE4 版本条件
        effective_ue4_version = summary.file_version_ue4 if not is_ue5_file else 1000

        # 检查是否需要读取条件字段
        read_package_name = effective_ue4_version >= VER_UE4_PACKAGESPLIT_PACKAGE_EXPORTS
        read_import_optional = is_ue5_file and summary.file_version_ue5 >= UE5_OPTIONAL_RESOURCES

        # IsFilterEditorOnly 检查（package_flags）
        is_filter_editor_only = (summary.package_flags & PKG_FilterEditorOnly) != 0

        for _ in range(summary.import_count):
            class_package = archive.read_name(name_map)
            class_name = archive.read_name(name_map)
            outer_index = PackageIndex(archive.read_i32())
            object_name = archive.read_name(name_map)

            # PackageName 条件字段
            package_name = None
            if read_package_name and not is_filter_editor_only:
                package_name = archive.read_name(name_map)

            # bImportOptional 条件字段
            b_import_optional = None
            if read_import_optional:
                b_import_optional = bool(archive.read_u8())

            import_map.append(ObjectImport(
                class_package=class_package,
                class_name=class_name,
                outer_index=outer_index,
                object_name=object_name,
                package_name=package_name,
                b_import_optional=b_import_optional
            ))

        return import_map
    ```

    关键修改：
    1. 添加 UE5 文件判断
    2. 添加 PackageName 条件读取（UEVer >= 518 且 !IsFilterEditorOnly）
    3. 添加 bImportOptional 条件读取（UEVer >= 1003）
    4. 更新 ObjectImport 构造包含新字段

        """测试 UE5 ImportMap 条件字段读取。"""

        data = bytearray()
