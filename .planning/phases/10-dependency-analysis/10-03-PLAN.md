    ```python
    def detect_circular_deps(import_map: List[ObjectImport]) -> List[List[str]]:
        """
        检测 ImportMap 中的潜在循环依赖（DEPS-03）。

        Per D-10-10: DFS 图遍历思想（简化实现）
        Per D-10-11: 检测同一 class_package 的多次引用（潜在循环）
        Per D-10-12: 路径数组格式 [pkg, pkg]

        注：ImportMap 仅包含"当前包→外部包"的单向依赖，
        无法确认跨包循环（需要分析外部包的 ImportMap）。
        此实现检测高密度依赖（同一包的多次引用）作为潜在循环警告。

        Args:
            import_map: 导入表列表

        Returns:
            List[List[str]]: 潜在循环路径列表，每个路径为 [package_name, package_name]
        """
        if not import_map:
            return []

        # 收集每个 class_package 的引用次数
        package_refs: Dict[str, int] = {}
        for imp in import_map:
            pkg = imp.class_package
            package_refs[pkg] = package_refs.get(pkg, 0) + 1

        # D-10-11: 检测同一包的多次引用（>1 次）→ 可能存在循环
        # D-10-12: 路径格式 [pkg, pkg]
        cycles = []
        for pkg, count in package_refs.items():
            if count > 1:
                # 多次引用同一包 → 添加潜在循环警告
                cycles.append([pkg, pkg])

        return cycles
    ```

    注意：
    - 这是一个简化实现（单文件分析无法确认跨包循环）
    - 使用 Dict[str, int] 统计引用次数
    - 返回格式为 List[List[str]]（每个循环为包名数组）
  </action>
