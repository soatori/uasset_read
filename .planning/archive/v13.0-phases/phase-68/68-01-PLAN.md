```python
供 N2CNodeTypeRegistry 和 N2CNodeType 枚举使用。
数据来源：UE5.8 Engine/Source/Editor/BlueprintGraph/Classes + AnimGraph + AIGraph
"""

# class_name → 语义类型枚举名映射（126 种）
# 特殊映射已在 RESEARCH.md 中标注（如 K2Node_IfThenElse → Branch）
K2NODE_ENUM_NAMES: dict[str, str] = {
    "K2Node_CallFunction": "CallFunction",
    "K2Node_CallArrayFunction": "CallArrayFunction",
    # ... 126 种映射
    "K2Node_IfThenElse": "Branch",           # 语义映射
    "K2Node_ExecutionSequence": "Sequence",  # 语义映射
    "K2Node_SwitchInteger": "SwitchInt",     # 语义映射

    "K2Node_CommutativeAssociativeBinaryOperator": "BinaryOperator",  # 缩短名
    "K2Node_ForEachElementInEnum": "ForEachEnum",  # 缩短名
    # ... 完整映射见 RESEARCH.md 表格
}

# class_name → parent_class_name 继承关系映射
K2NODE_INHERITANCE: dict[str, str] = {
    "K2Node_CallArrayFunction": "K2Node_CallFunction",
    "K2Node_CallDataTableFunction": "K2Node_CallFunction",
    "K2Node_CallDelegate": "K2Node_CallFunction",
    "K2Node_CallFunctionOnMember": "K2Node_CallFunction",
    "K2Node_CallMaterialParameterCollectionFunction": "K2Node_CallFunction",
    "K2Node_CallParentFunction": "K2Node_CallFunction",
    "K2Node_Message": "K2Node_CallFunction",
    "K2Node_GetInputAxisValue": "K2Node_CallFunction",
    "K2Node_GetInputAxisKeyValue": "K2Node_CallFunction",
    "K2Node_GetInputVectorAxisValue": "K2Node_CallFunction",
    "K2Node_PromotableOperator": "K2Node_CallFunction",
    "K2Node_CommutativeAssociativeBinaryOperator": "K2Node_CallFunction",
    "K2Node_InstancedStruct": "K2Node_CallFunction",
    "K2Node_AddComponent": "K2Node_CallFunction",
    "K2Node_VariableGet": "K2Node_Variable",
    "K2Node_VariableSet": "K2Node_Variable",
    "K2Node_VariableSetRef": "K2Node_Variable",
    "K2Node_StructMemberGet": "K2Node_StructOperation",
    "K2Node_StructMemberSet": "K2Node_StructOperation",
    "K2Node_SetFieldsInStruct": "K2Node_StructOperation",
    "K2Node_Event": "K2Node_EditablePinBase",
    "K2Node_ActorBoundEvent": "K2Node_Event",
    # ... 完整继承链见 RESEARCH.md Section "继承层次结构"
}

# 所有类型列表（从 K2NODE_ENUM_NAMES.keys() 派生）
K2NODE_TYPES: list[str] = list(K2NODE_ENUM_NAMES.keys())


def get_parent_chain(class_name: str) -> list[str]:
    """获取完整继承链（从 class_name 到根）。"""
    chain = [class_name]
    current = class_name
    visited = set()
    while current in K2NODE_INHERITANCE:
        if current in visited:
            break  # 循环保护
        visited.add(current)
        parent = K2NODE_INHERITANCE[current]
        chain.append(parent)
        current = parent
    return chain
```

- 映射数据来自 RESEARCH.md 的完整 126 种类型清单
