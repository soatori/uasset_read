class FPackageIndex {
    int32 Index;
    bool IsImport() const { return Index < 0; }
    bool IsExport() const { return Index > 0; }
    bool IsNull() const { return Index == 0; }
    int32 ToImport() const { return -Index - 1; }
    int32 ToExport() const { return Index - 1; }
    static FPackageIndex FromImport(int32 ImportIndex) { return FPackageIndex(-ImportIndex - 1); }
    static FPackageIndex FromExport(int32 ExportIndex) { return FPackageIndex(ExportIndex + 1); }
};
