<objective>
修复 UAT 测试真实 Lyra UE5 文件时发现的 SavedHash 解析 bug。

输出：修正后的 PackageFileSummary dataclass 和 read_package_summary()，正确处理 PACKAGE_SAVED_HASH 版本阈值。
</objective>

    ```python
    def test_saved_hash_ue5_package_saved_hash_version():
        """测试 UE5 >= PACKAGE_SAVED_HASH (1004) 的 SavedHash 和早期 TotalHeaderSize 解析。"""
        # 创建 UE5 版本 1004 (PACKAGE_SAVED_HASH) 的合成数据
        data = bytearray()
        
        # 按正确顺序的文件头字段：
        # 1. Magic tag (0x9E2A83C1)
        data.extend(struct.pack('<I', 0x9E2A83C1))
        
        # 2. LegacyFileVersion (-8 for UE5)
        data.extend(struct.pack('<i', -8))
        
        # 3. LegacyUE3Version (skipped for legacy=-8)
        
        # 4. UE4 version
        data.extend(struct.pack('<i', 1000))
        
        # 5. UE5 version (1004 = PACKAGE_SAVED_HASH)
        data.extend(struct.pack('<i', 1004))
        
        # 6. LicenseeVersion
        data.extend(struct.pack('<i', 0))
        
        # 7. SavedHash (20 bytes) - UE5 >= 1004 新增
        test_saved_hash = bytes([0xAA] * 20)  # Test pattern
        data.extend(test_saved_hash)
        
        # 8. TotalHeaderSize (early read) - UE5 >= 1004 新增
        data.extend(struct.pack('<i', 256))
        
        # 9. CustomVersions (count = 0)
        data.extend(struct.pack('<I', 0))
        
        # 10. PackageFlags
        data.extend(struct.pack('<I', 0))
        
        # 11. NameCount + NameOffset
        data.extend(struct.pack('<i', 0))  # NameCount
        data.extend(struct.pack('<i', len(data) + 8))  # NameOffset
        
        # Pad to reasonable size
        while len(data) < 300:
            data.append(0)
        
        # Write and parse
        with tempfile.NamedTemporaryFile(suffix='.uasset', delete=False) as f:
            f.write(data)
            temp_path = f.name
        
        try:
            result = parse_uasset(temp_path)
            assert result.is_success, f"Parse failed: {result.errors}"
            assert result.summary.saved_hash == test_saved_hash, \
                f"SavedHash mismatch: expected {test_saved_hash.hex()}, got {result.summary.saved_hash.hex()}"
            assert result.summary.total_header_size == 256, \
                f"TotalHeaderSize mismatch: expected 256, got {result.summary.total_header_size}"
        finally:
            os.unlink(temp_path)
    ```

    此测试确保解析器正确处理 UE5 >= 1004 文件的 SavedHash/TotalHeaderSize 字段。
