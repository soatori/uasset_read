def _parse_property_flags(self, property_flags: int) -> dict:

### 步骤 4: 更新变量解析逻辑

**文件**: `src/core/archive.py`

```python
def read_blueprint_variables(self, blueprint_class: ObjectExport) -> List[BlueprintVariable]:
    """读取蓝图变量"""
    variables = []
    
    # 遍历 Blueprint 的 NewVariables
    if blueprint_class and hasattr(blueprint_class, 'new_variables'):
        for var_export in blueprint_class.new_variables:
            # 读取变量名称
            var_name = self.get_name(var_export.name_index)
            
            # 读取变量类型
            var_type = self.get_property_type(var_export)
            
            # 读取默认值
            default_value = self.get_default_value(var_export)
            
            # 读取属性标志
            property_flags = var_export.property_flags
            
            # 读取元数据
            meta_data = self.read_metadata(var_export)
            
            # 读取分类
            category = meta_data.get('Category', '')
            edit_category = meta_data.get('EditCondition', '')
            
            # 解析属性标志
            flags = self._parse_property_flags(property_flags)
            
            variables.append(BlueprintVariable(
                name=var_name,
                var_type=var_type,
                default_value=default_value,
                property_flags=property_flags,
                category=category,
                edit_condition=edit_condition,
                meta_data=meta_data,
                **flags
            ))
    
    return variables
```
