```python
def format_as_json(result: ParseResult, indent: int = 2, include_metadata: bool = True) -> str:
    """格式化为 JSON（支持增强元数据）"""
    output = {
        "version": SCHEMA_VERSION,
        "status": _get_status(result)
    }
    
    if result.blueprint:
        blueprint_data = {
            "name": result.blueprint.name,
            "parent_class": result.blueprint.parent_class
        }
        
        # 增强的变量输出
        if hasattr(result.blueprint, 'variables') and result.blueprint.variables:
            blueprint_data["variables"] = [_format_variable_enhanced(var) for var in result.blueprint.variables]
        
        # 增强的函数输出
        if hasattr(result.blueprint, 'functions') and result.blueprint.functions:
            blueprint_data["functions"] = [_format_function_enhanced(func) for func in result.blueprint.functions]
        
        # 增强的事件输出
        if hasattr(result.blueprint, 'events') and result.blueprint.events:
            blueprint_data["events"] = [_format_event_enhanced(event) for event in result.blueprint.events]
        
        output["blueprint"] = blueprint_data
    
    if result.graphs:
        output["graphs"] = [_format_graph(graph) for graph in result.graphs]
    
    return json.dumps(output, indent=indent, ensure_ascii=False, default=_json_serializer)
```
