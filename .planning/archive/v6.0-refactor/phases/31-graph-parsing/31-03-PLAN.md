```python
"""蓝图图解析入口 — 从 ExportMap 提取所有 EdGraph/UberEdGraph。

等价迁移 uasset_read.py L3095-3143。
Phase 31: 蓝图图解析模块。
"""
from __future__ import annotations

from typing import TYPE_CHECKING, List

if TYPE_CHECKING:
    from uasset_read.archive import FArchive
    from uasset_read.serializers.package_summary import PackageFileSummary
    from uasset_read.serializers.object_resources import ObjectExport, ObjectImport

from uasset_read.constants import PKG_Cooked
from uasset_read.serializers.object_resources import get_asset_class
from uasset_read.serializers.graph import read_ue_graph
from uasset_read.models.core import UEdGraph


def extract_blueprint_graphs(
    archive: FArchive,
    summary: PackageFileSummary,
    name_map: List[str],
    import_map: List[ObjectImport],
    export_map: List[ObjectExport],
) -> List[UEdGraph]:
    """
    从 ExportMap 提取蓝图图（等价迁移 uasset_read.py L3095-3143）。

    遍历 ExportMap，ClassIndex 解析后包含 "EdGraph" 或 "UberEdGraph" 的导出视为图对象。
    对每个图调用 read_ue_graph 完整解析 Graph→Node→Pin 三层结构。

    安全检查：PKG_Cooked 检查避免解析已剥离资产。

    Returns:
        List[UEdGraph]: 检测到的图列表
    """
    graphs: List[UEdGraph] = []

    # PKG_Cooked 检查 — cooked 资产无图数据
    is_cooked = (summary.package_flags & PKG_Cooked) != 0
    if is_cooked:
        return []

    # 遍历 ExportMap 寻找 EdGraph/UberEdGraph 类型导出
    for export_idx, export in enumerate(export_map):
        class_name = get_asset_class(export, import_map, export_map)

        if class_name and class_name in ('EdGraph', 'UberEdGraph'):
            graph = read_ue_graph(
                archive, name_map, summary,
                export_map, import_map,
                export, class_name, export_idx + 1  # 1-based index
            )
            graphs.append(graph)

    return graphs
```

**辅助函数 1: `_derive_node_name(node: UEdGraphNode, idx: int) -> str`**
- 如果 node.class_name == "K2Node_CallFunction" 且 node.node_data.function_reference.member_name 非空：返回 member_name
- 如果 node.class_name == "K2Node_Event" 且 node.node_data.event_reference.member_name 非空：返回 member_name
- 否则返回 f"{node.class_name}_{idx}"

**辅助函数 2: `format_pin_ref(node_guid, pin_name, node_name_lookup, mode) -> str | dict`**
- mode == "name": 返回 f"{node_name_lookup.get(node_guid, '?')}:{pin_name}"
- mode == "guid": 返回 {"node_guid": node_guid, "pin_name": pin_name}
- 从 constants.FORMAT_CONFIG["pin_reference_mode"] 获取 mode

**辅助函数 3: `_get_start_event_name(start_node) -> str`**
- K2Node_Event: 返回 event_reference.member_name 或 class_name
- K2Node_EnhancedInputAction: 返回 class_name
- K2Node_VariableSet: 返回 class_name
- K2Node_CustomEvent: 返回 class_name

**辅助函数 4: `_find_next_exec_node(node, pin_lookup, node_lookup) -> Optional[UEdGraphNode]`**
- 遍历 node.pins, 找 direction == 1 (Output) 且 pin_type.pin_category == "exec" 的 pin
- 对该 pin 的 linked_to_raw 中每个 linked_pin_id:
  - target_pin_guid = linked_pin_id.get("pin_guid") if isinstance(linked_pin_id, dict) else linked_pin_id
  - 如果 target_pin_guid in pin_lookup: 返回 node_lookup.get(target_node_guid)
- 返回 None

**辅助函数 5: `_trace_execution_from_event(start_node, pin_lookup, node_lookup) -> List[Dict]`**
- visited = set(), flow = [], current_node = start_node
- while current_node:
  - 循环检测: if current_node.node_guid in visited → 添加 {"node_guid": ..., "node_type": ..., "cycle_detected": True}, break
  - visited.add(current_node.node_guid)
  - node_info = {"node_guid": current_node.node_guid, "node_type": current_node.class_name}
  - 如果 class_name == "K2Node_CallFunction" 且有 function_reference: 添加 "function_name" = function_reference.member_name
  - 如果 class_name == "K2Node_Event" 且有 event_reference: 添加 "event_name" = event_reference.member_name
  - 如果 class_name in CONTROL_FLOW_NODES: 添加 "branch_type" = BRANCH_TYPE_MAP.get(class_name, "unknown"), "stopped_at" = "control_flow_node", 添加后 break
  - flow.append(node_info)
  - current_node = _find_next_exec_node(current_node, pin_lookup, node_lookup)
- 返回 flow

**辅助函数 6: `_trace_execution_from_pin(start_node, start_pin, pin_lookup, node_lookup) -> List[Dict]`**
- 类似 _trace_execution_from_event，但起点是指定的 start_pin 而非遍历所有 exec output

**主函数 1: `build_connections_map(graph: UEdGraph) -> Tuple[List[Dict], List[str]]`**
- 构建 node_name_lookup: {node_guid: _derive_node_name(node, idx)}
- 构建 pin_lookup: {pin_id: (node_guid, pin_name)}
- mode = FORMAT_CONFIG["pin_reference_mode"]
- connections = [], warnings = []
- 遍历所有 nodes → pins → if pin.direction == 1 (Output):
  - 对 linked_to_raw 中每个 target:
    - target_pin_guid = target.get("pin_guid") if isinstance(target, dict) else target
    - 如果在 pin_lookup 中找到: connections.append({"from": format_pin_ref(...), "to": format_pin_ref(...)})
    - 否则: warnings.append(f"PinId {target_pin_guid} not found in graph")
- 返回 (connections, warnings)

**主函数 2: `build_execution_flows(graph: UEdGraph) -> List[Dict]`**
- 构建 pin_lookup 和 node_lookup
- execution_flows = []
- start_nodes = [n for n in graph.nodes if n.class_name in START_EVENT_TYPES]
- 对每个 start_node:
  - 如果 class_name == "K2Node_EnhancedInputAction":
    - 遍历 output exec pins (direction == 1 and pin_type.pin_category == "exec"):
      - flow = _trace_execution_from_pin(start_node, pin, pin_lookup, node_lookup)
      - execution_flows.append({"start_event": f"{start_node.class_name}.{pin.pin_name}", "nodes": flow})
  - 否则:
    - flow = _trace_execution_from_event(start_node, pin_lookup, node_lookup)
    - execution_flows.append({"start_event": _get_start_event_name(start_node), "nodes": flow})
- 返回 execution_flows

**主函数 3: `build_data_flows(graph: UEdGraph, mode: str = "name") -> List[Dict]`**
- 构建 pin_lookup 和 node_name_lookup
- data_flows = []
- 遍历所有 nodes → pins:
  - if pin.direction == 1 and pin.pin_type and pin.pin_type.pin_category != "exec":
    - 对 linked_to_raw 中每个 target:
      - target_pin_guid = target.get("pin_guid") if isinstance(target, dict) else target
      - 如果在 pin_lookup 中找到: data_flows.append({"source": format_pin_ref(...), "target": format_pin_ref(...)})
- 返回 data_flows
