```python
def test_ue57_bool_serialization():
    """测试 UE 5.7 资产的 Bool 序列化修复。"""
    ue57_asset_path = "E:/Develop/lib/UnrealEngine/Samples/FirstPerson/Content/FirstPerson/Blueprints/BP_FirstPersonCharacter.uasset"
    
    if not os.path.exists(ue57_asset_path):
        pytest.skip(f"UE 5.7 test asset not found: {ue57_asset_path}")
    
    result = parse_uasset(ue57_asset_path)
    
    assert result.status == "success"
    assert result.errors_count == 0
    assert len(result.export_map) > 0
    
    file_size = 138384
    for i, export in enumerate(result.export_map):
        if export.serial_offset is not None:
            assert 0 <= export.serial_offset <= file_size
    
    has_blueprint = any("BP_FirstPersonCharacter" in e.object_name for e in result.export_map)
    assert has_blueprint


def test_read_bool_method():
    """测试 FArchive.read_bool() 方法的行为。"""
    import struct
    import tempfile
    
    true_data = struct.pack('<I', 1)
    false_data = struct.pack('<I', 0)
    
    with tempfile.NamedTemporaryFile(delete=False) as f:
        f.write(true_data + false_data)
        temp_path = f.name
    
    try:
        archive = FArchive(temp_path)
        assert archive.read_bool() == True
        assert archive.read_bool() == False
    finally:
        os.unlink(temp_path)
```
