# tests/test_quality_stats.py
"""quality_stats.py 脚本功能验证测试。"""
import subprocess
import sys
import json
import os
import tempfile


def _run_quality_stats(args=None):
    """运行 quality_stats.py 并返回 (returncode, stdout, stderr)。"""
    cmd = [sys.executable, "scripts/quality_stats.py"]
    if args:
        cmd.extend(args)
    result = subprocess.run(cmd, capture_output=True, text=True)
    return result.returncode, result.stdout, result.stderr


class TestQualityStatsBasic:
    """基础功能测试。"""

    def test_help_exits_clean(self):
        rc, out, err = _run_quality_stats(["--help"])
        assert rc == 0
        assert "scan_dir" in out or "usage" in out.lower()

    def test_missing_scan_dir_errors(self):
        rc, out, err = _run_quality_stats()
        assert rc != 0


class TestQualityStatsMetrics:
    """指标检测测试 — 使用合成 C++ 文件。"""

    def _make_cpp_file(self, tmp_path, content, name="test.cpp"):
        path = tmp_path / name
        path.write_text(content)
        return str(tmp_path)

    def test_detects_function_placeholder(self, tmp_path):
        """Function_N 占位符应被检测。"""
        cpp_dir = self._make_cpp_file(tmp_path, """
void UTestClass::SomeFunction()
{
    CallFunction_42(arg1);
    CallFunction_99(arg2);
}
""")
        rc, out, err = _run_quality_stats([cpp_dir])
        assert rc == 0
        assert "Function_N" in out or "placeholder" in out.lower() or "占位" in out

    def test_detects_goto_fallback(self, tmp_path):
        """goto Label_ 回退应被检测。"""
        cpp_dir = self._make_cpp_file(tmp_path, """
void UTestClass::ExecuteUbergraph()
{
    goto Label_1;
Label_1:
    goto Label_2;
Label_2:
    return;
}
""")
        rc, out, err = _run_quality_stats([cpp_dir])
        assert rc == 0
        assert "goto" in out.lower()

    def test_json_output_valid(self, tmp_path):
        """--json 输出应为有效 JSON。"""
        cpp_dir = self._make_cpp_file(tmp_path, """
void UTestClass::Test()
{
    return;
}
""")
        rc, out, err = _run_quality_stats([cpp_dir, "--json"])
        assert rc == 0
        data = json.loads(out)  # 不应抛异常
        assert isinstance(data, dict)

    def test_threshold_flag_custom(self, tmp_path):
        """--threshold 应使用自定义阈值。"""
        cpp_dir = self._make_cpp_file(tmp_path, """
void UTestClass::Test() { return; }
""")
        rc, out, err = _run_quality_stats([cpp_dir, "--threshold", "5"])
        assert rc == 0

    def test_verbose_flag_output(self, tmp_path):
        """--verbose 应输出每个文件详细统计。"""
        cpp_dir = self._make_cpp_file(tmp_path, """
void UTestClass::Test() { return; }
""")
        rc, out, err = _run_quality_stats([cpp_dir, "--verbose"])
        assert rc == 0
        # verbose 模式应比非 verbose 输出更多内容
        assert len(out) > 0

# tests/test_ue4_legacy_xfail.py
"""UE4 legacy_file_version=-3 资产的 xfail 策略测试。

验证：
1. legacy=-3 资产被正确识别为 UE4 资产
2. 抛出明确的 UnsupportedVersionError（不是 VersionError 通用消息）
3. 测试标记为 xfail 且有明确的 reason
"""
import pytest

from uasset_read.serializers.package_summary import (
    UE5_LEGACY_VERSIONS,
    read_package_summary,
)


class TestUE5LegacyVersions:
    """验证 UE5 支持版本列表正确。"""

    def test_ue5_supported_legacy_versions(self):
        """UE5_LEGACY_VERSIONS 应包含 -9 和 -8。"""
        assert -9 in UE5_LEGACY_VERSIONS
        assert -8 in UE5_LEGACY_VERSIONS

    def test_ue4_legacy_not_supported(self):
        """UE4 legacy 版本 (-7 到 -3) 默认不支持。"""
        for v in range(-7, -2):
            assert v not in UE5_LEGACY_VERSIONS, (
                f"legacy_version={v} should not be in UE5_LEGACY_VERSIONS "
                "unless explicitly supported"
            )


class TestUE4LegacyVersionError:
    """验证 UE4 资产产生明确的版本错误。"""

    @pytest.fixture
    def ue4_asset_path(self, tmp_path):
        """创建模拟 UE4 legacy=-3 的 .uasset 文件。"""
        path = tmp_path / "ue4_legacy.uasset"
        # UE magic + 版本号 + legacy=-3
        data = (
            b"\xC1\x83\x2A\x9E"  # magic
            + b"\x00" * 8         # 填充
            + b"\xFD\xFF\xFF\xFF"  # legacy_file_version = -3 (little-endian i32)
            + b"\x00" * 100       # 剩余头部
        )
        path.write_bytes(data)
        return str(path)

    def test_ue4_legacy_raises_clear_error(self, ue4_asset_path):
        """UE4 资产应产生明确的版本不支持错误。"""
        from uasset_read.archive import FArchive
        from uasset_read.serializers.package_summary import read_package_summary

        with open(ue4_asset_path, "rb") as f:
            archive = FArchive(f)
            with pytest.raises(Exception) as exc_info:
                read_package_summary(archive)

        error_msg = str(exc_info.value).lower()
        # 错误消息应提及 UE4 或版本信息
        assert "ue4" in error_msg or "legacy" in error_msg or "version" in error_msg


class TestPFireAssetStrategy:
    """P_Fire.uasset 的特殊处理策略。"""

    @pytest.mark.xfail(
        reason="P_Fire.uasset 使用 UE4 legacy_file_version=-3，暂不支持"

        with pytest.raises(Exception):
            parse_single(p_fire_path, tolerant=False)
